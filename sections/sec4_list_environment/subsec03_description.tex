\begin{frame}
\frametitle{描述列表}
\framesubtitle{description 环境}

\begin{block}{核心组件}
\begin{description}
\item[Beamer] LaTeX 演示文稿标准类，提供丰富的幻灯片功能
\item[ctex] 中文排版宏包，支持中英文混排和字体切换
\item[TikZ] 强大的矢量绘图工具，可直接在 LaTeX 中绘制图形
\item[biblatex] 现代化的参考文献管理方案，支持多种引用格式
\end{description}
\end{block}

\begin{exampleblock}{编译工具链}
\begin{description}
\item[XeLaTeX] 支持 Unicode 和系统字体的 LaTeX 编译器
\item[biber] 与 biblatex 配合使用的参考文献处理工具
\item[Make] 自动化构建工具，通过 Makefile 简化编译流程
\end{description}
\end{exampleblock}
\end{frame}
