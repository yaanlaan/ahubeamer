\begin{alertblock}{注意事项}
\begin{enumerate}
  \item 编译时需执行 \texttt{biber} 处理参考文献
  \item 引用标签需与 \texttt{.bib} 文件中一致
  \item 多次编译以确保引用标签正确解析
\end{enumerate}
\end{alertblock}
