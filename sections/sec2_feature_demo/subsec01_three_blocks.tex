\begin{frame}
\frametitle{三种块环境}
\framesubtitle{普通块 · 警告块 · 示例块}

\begin{block}{普通块 (block)}
用于常规内容展示，标题为深蓝底白字，正文为浅蓝底黑字。
这是最常用的块环境，适合放置一般性说明文字。
\end{block}

\begin{alertblock}{警告块 (alertblock)}
用于强调重要信息或警告提示，标题为红底白字。
\textbf{注意：} 请谨慎使用，避免过度强调。
\end{alertblock}

\begin{exampleblock}{示例块 (exampleblock)}
用于展示示例或正面信息，标题为绿底白字。
\textit{提示：} 适合放置代码示例、成功案例等。
\end{exampleblock}
\end{frame}
