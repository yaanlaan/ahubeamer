\begin{frame}
\frametitle{关于本模板}
\framesubtitle{AHU Beamer Theme}

\begin{block}{模板特性}
\begin{itemize}
  \item 基于 \textbf{beamer} 的安徽大学演示文稿模板
  \item 蓝白配色方案，与校徽颜色统一
  \item 支持 \alert{中文} 排版（ctex 宏包）
  \item 三种块环境：\structure{普通块}、\alert{警告块}、\textbf{示例块}
  \item 三级列表样式，颜色主题贯穿始终
\end{itemize}
\end{block}

\begin{exampleblock}{技术栈}
\begin{itemize}
  \item 编译器：\texttt{XeLaTeX} + \texttt{biber}
  \item 字体：微软雅黑 / SimHei / Arial / Times New Roman
  \item 绘图：TikZ（含 3D、矩阵、箭头等库）
  \item 参考文献：biblatex + biber，国标格式
\end{itemize}
\end{exampleblock}
\end{frame}
